%----------CUSTOM COMMANDS----------
% \unskip drops the trailing space of #1; without it, a bullet that exactly
% fills its last line breaks at that space and adds an empty line.
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1\unskip\vspace{-2pt}}
  }
}

% One full-width row: #1 flush left, #2 flush right. Every heading below
% is built from it.
\newcommand{\resumeRow}[2]{%
  \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
    #1 & #2 \\
  \end{tabular*}\vspace{-7pt}%
}

% True right after an org or title row, false after a bullet list.
\newif\ifresumeafterheading

% List spacing saved while a stacked \resumeTitle uses a rigid copy.
\newskip\resumeSavedItemsep
\newskip\resumeSavedParskip

% Bold organization (employer or university) and its location.
\newcommand{\resumeOrg}[2]{%
  \vspace{-2pt}\item
  \resumeRow{\textbf{#1}}{#2}%
  \global\resumeafterheadingtrue
}

% Italic title (job title or degree) and its dates. Repeat it for
% promotions; bullets may follow any title. Stacked under a row, a title
% keeps rigid spacing like the rows of one table (1\skip drops a skip's
% stretch and shrink); after bullets it gets the usual flexible gap.
\newcommand{\resumeTitle}[2]{%
  \vspace{-2pt}%
  \ifresumeafterheading
    \resumeSavedItemsep=\itemsep \resumeSavedParskip=\parskip
    \itemsep=1\itemsep\relax \parskip=1\parskip\relax
  \fi
  \item
  \resumeRow{\textit{\small#1}}{\textit{\small #2}}%
  \ifresumeafterheading
    \itemsep=\resumeSavedItemsep \parskip=\resumeSavedParskip
  \fi
  \global\resumeafterheadingtrue
}

\newcommand{\resumeProjectHeading}[2]{%
  \item
  \resumeRow{\small#1}{#2}%
}
